# Q3

python
sensors = {'S1': 22.1, 'S2': 19.8, 'S3': 25.4}

sensor_status = {key: ('High' if value > 20 else 'Low') for key, value in sensors.items()}

print(sensor_status)

Output:
{'S1': 'High', 'S2': 'Low', 'S3': 'High'}

## Q4

python
def safe_convert(value):
    try:
        return int(value)
    except (ValueError, TypeError):
        return None

result1 = safe_convert('10')
result2 = safe_convert('missing')

print(result1)
print(result2)

Output:
10
None

## Q5

python
def safe_convert(value):
    try:
        return int(value)
    except (ValueError, TypeError):
        return None

data = [{'id': 1, 'val': '10'}, {'id': 2, 'val': 'bad'}, {'id': 3, 'val': '30'}]

converted_values = [safe_convert(record['val']) for record in data]

print(converted_values)

Output:
[10, None, 30]
